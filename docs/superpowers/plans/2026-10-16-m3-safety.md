# M3「安全」Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 给 Agent 补上安全边界——工具权限三级（`safe`/`confirm`/`blocked`）、`confirm` 级工具的网页确认卡片闭环、`code.exec` 子进程代码沙箱。

**Architecture:** 沿用 M1/M2 三层解耦与 SSE 契约。runtime 新增内存确认闸门（`ConfirmBroker`）与 `POST /runs/:id/confirm`；`agent-loop` 遇 `confirm` 工具先 `yield confirm_request` 再 `await` 闸门；BFF 增 `POST /api/runs/:id/confirm`（JWT + 归属校验 + 转发）；web 识别 `confirm_request` 事件渲染内联卡片。`code.exec` 走「子进程 + Node `--permission` + 受限 harness（无 fs/网络/子进程）」。

**Tech Stack:** TypeScript 5.6 · Node 22（`--permission`）· pnpm + Turborepo · NestJS 10 (Fastify) · Prisma + Postgres · Next.js 15 / React 19 · vitest

**Spec:** `docs/superpowers/specs/2026-10-16-m3-safety-design.md`

```ts
export type ConfirmReason = 'user_denied' | 'confirm_timeout';

export interface ConfirmDecisionResult {
  approved: boolean;
  reason?: ConfirmReason;
}

/**
 * 确认闸门：`confirm` 级工具在执行前挂起，等用户决策或超时。
 * 单实例内存态（与 v1 无状态单实例现状匹配）；跨重启恢复是 M5 队列化的活。
 */
export interface ConfirmGate {
  /** 挂起等待决策；超时按 approved=false + reason=confirm_timeout 兑现。同一 runId 至多一个挂起。 */
  request(runId: string, confirmId: string, timeoutMs: number): Promise<ConfirmDecisionResult>;
  /** 兑现挂起闸门；confirmId 不匹配或不存在返回 false（幂等，不抛错）。 */
  resolve(runId: string, confirmId: string, approved: boolean): boolean;
  /** 连接断开兜底：按拒绝兑现并清理。 */
  cancel(runId: string): void;
}

interface Pending {
  confirmId: string;
  settle: (r: ConfirmDecisionResult) => void;
  timer: NodeJS.Timeout;
}

export class ConfirmBroker implements ConfirmGate {
  private readonly pending = new Map<string, Pending>();

  request(runId: string, confirmId: string, timeoutMs: number): Promise<ConfirmDecisionResult> {
    return new Promise((resolveResult) => {
      if (this.pending.has(runId)) {
        // 工具调用串行，理论不会并发；防御性拒绝新请求，避免状态错乱。
        resolveResult({ approved: false });
        return;
      }
      const timer = setTimeout(() => {
        const p = this.pending.get(runId);
        if (!p) return;
        this.pending.delete(runId);
        p.settle({ approved: false, reason: 'confirm_timeout' });
      }, timeoutMs);
      timer.unref?.();
      this.pending.set(runId, { confirmId, settle: resolveResult, timer });
    });
  }

  resolve(runId: string, confirmId: string, approved: boolean): boolean {
    const p = this.pending.get(runId);
    if (!p || p.confirmId !== confirmId) return false;
    this.pending.delete(runId);
    clearTimeout(p.timer);
    p.settle(approved ? { approved: true } : { approved: false, reason: 'user_denied' });
    return true;
  }

  cancel(runId: string): void {
    const p = this.pending.get(runId);
    if (!p) return;
    this.pending.delete(runId);
    clearTimeout(p.timer);
    p.settle({ approved: false, reason: 'user_denied' });
  }
}
```

```ts
/** 第一轮发起一次工具调用、第二轮收尾的桩网关。 */
function oneShotGateway(call: {
  id: string;
  name: string;
  args: Record<string, unknown>;
}): ModelGateway {
  let round = 0;
  return {
    async chat() {
      round++;
      if (round === 1) {
        return { content: '执行', toolCalls: [call], usage: { input: 0, output: 0 } };
      }
      return { content: '收尾', toolCalls: [], usage: { input: 0, output: 0 } };
    },
  };
}

/** 立即兑现的桩闸门，避免测试里真的挂起。 */
function stubGate(approved: boolean, reason?: 'user_denied' | 'confirm_timeout'): ConfirmGate {
  return {
    request: async () =>
      approved ? { approved: true } : { approved: false, reason: reason ?? 'user_denied' },
    resolve: () => false,
    cancel: () => {},
  };
}
```

```ts
import { spawn as nodeSpawn } from 'node:child_process';
import type { Tool } from './types';

export interface CodeExecResult {
  ok: boolean;
  stdout: string;
  stderr: string;
  durationMs: number;
  timeout?: boolean;
  truncated?: boolean;
}

type Spawn = typeof nodeSpawn;

interface Limits {
  timeoutMs: number;
  maxOldSpaceMb: number;
  maxOutputBytes: number;
}

/**
 * 受限 harness：读 stdin 取用户代码，在**无 require/process/网络**的作用域内执行，结果以 JSON 写 stdout。
 * 用 String.raw 保证 `\n` 原样进入子进程源码；harness 内不得出现 `${}`（会被模板插值）。
 * 用 `-e` 内联而非独立文件：runtime 构建为 tsup 单入口，附带的 .js 资产不会被拷进 dist。
 */
const HARNESS = String.raw`
const vm = require('node:vm');
const MAX_BYTES = Number(process.argv[1] || 65536);
const fmt = (x) => {
  if (typeof x === 'string') return x;
  try { const s = JSON.stringify(x); return s === undefined ? String(x) : s; } catch { return String(x); }
};
let input = '';
process.stdin.setEncoding('utf8');
process.stdin.on('data', (d) => { input += d; });
process.stdin.on('end', () => {
  const out = [];
  let outLen = 0;
  const print = function () {
    const line = Array.prototype.map.call(arguments, fmt).join(' ');
    outLen += line.length + 1;
    if (outLen > MAX_BYTES) throw new Error('output_limit_exceeded');
    out.push(line);
  };
  // 受限作用域：仅暴露纯计算原语与 print，不暴露 require/process/fetch/网络。
  const sandbox = { print };
  let payload;
  try {
    const result = vm.runInNewContext(input, sandbox);
    payload = { ok: true, stdout: out.join('\n'), result: result === undefined ? null : fmt(result) };
  } catch (e) {
    payload = { ok: false, stdout: out.join('\n'), stderr: String((e && e.message) || e) };
  }
  process.stdout.write(JSON.stringify(payload));
});
`;

interface HarnessOutput {
  ok: boolean;
  stdout: string;
  stderr: string;
}

function parseHarnessOutput(text: string): HarnessOutput | undefined {
  try {
    const v = JSON.parse(text) as { ok?: unknown; stdout?: unknown; stderr?: unknown };
    if (typeof v.ok !== 'boolean') return undefined;
    return {
      ok: v.ok,
      stdout: typeof v.stdout === 'string' ? v.stdout : '',
      stderr: typeof v.stderr === 'string' ? v.stderr : '',
    };
  } catch {
    return undefined;
  }
}

/** 在子进程沙箱内执行代码；返回结构化结果，**绝不抛出**（工具层只做透传）。 */
function runSandboxed(spawn: Spawn, code: string, limits: Limits): Promise<CodeExecResult> {
  return new Promise((resolve) => {
    const started = Date.now();
    let settled = false;
    let timedOut = false;
    let truncated = false;
    let outBytes = 0;
    let stdout = '';
    let stderr = '';

    const child = spawn(
      process.execPath,
      [
        '--permission',
        `--max-old-space-size=${limits.maxOldSpaceMb}`,
        '-e',
        HARNESS,
        String(limits.maxOutputBytes),
      ],
      { stdio: ['pipe', 'pipe', 'pipe'] },
    );

    const finish = (r: Omit<CodeExecResult, 'durationMs'>): void => {
      if (settled) return;
      settled = true;
      clearTimeout(timer);
      resolve({ ...r, durationMs: Date.now() - started });
    };

    const timer = setTimeout(() => {
      timedOut = true;
      child.kill('SIGKILL');
    }, limits.timeoutMs);

    const collect = (into: 'out' | 'err') => (chunk: Buffer): void => {
      outBytes += chunk.length;
      if (outBytes > limits.maxOutputBytes) {
        truncated = true;
        child.kill('SIGKILL');
        return;
      }
      if (into === 'out') stdout += chunk.toString('utf8');
      else stderr += chunk.toString('utf8');
    };
    child.stdout?.on('data', collect('out'));
    child.stderr?.on('data', collect('err'));

    child.on('error', (err) => finish({ ok: false, stdout, stderr: String(err) }));
    child.on('close', () => {
      if (timedOut) return finish({ ok: false, stdout, stderr, timeout: true });
      if (truncated) return finish({ ok: false, stdout, stderr, truncated: true });
      const parsed = parseHarnessOutput(stdout);
      if (parsed) return finish({ ok: parsed.ok, stdout: parsed.stdout, stderr: parsed.stderr });
      return finish({ ok: false, stdout, stderr });
    });

    // 代码经 stdin 送入：避免为代码文件开 fs 读权限。
    child.stdin?.write(code);
    child.stdin?.end();
  });
}

export function createCodeExecTool(
  env: NodeJS.ProcessEnv = process.env,
  spawn: Spawn = nodeSpawn,
): Tool {
  const limits: Limits = {
    timeoutMs: Number(env.CODE_EXEC_TIMEOUT_MS ?? 5000),
    maxOldSpaceMb: Number(env.CODE_EXEC_MAX_OLD_SPACE_MB ?? 128),
    maxOutputBytes: Number(env.CODE_EXEC_MAX_OUTPUT_BYTES ?? 65536),
  };
  return {
    name: 'code.exec',
    description:
      '在受限沙箱中执行一段 JavaScript 代码（纯计算：无文件/网络/子进程），用 print() 输出结果。需用户确认后执行。',
    permission: 'confirm',
    parameters: {
      type: 'object',
      properties: {
        code: { type: 'string', description: '要执行的 JavaScript 代码；用 print(...) 输出' },
        timeoutMs: { type: 'number', description: '可选：本次执行的墙钟上限（毫秒）' },
      },
      required: ['code'],
    },
    async execute(args) {
      const code = String(args.code ?? '');
      const requested = args.timeoutMs;
      // 调用方只能收紧、不能放宽服务端上限。
      const limit =
        typeof requested === 'number' && requested > 0
          ? Math.min(requested, limits.timeoutMs)
          : limits.timeoutMs;
      return runSandboxed(spawn, code, { ...limits, timeoutMs: limit });
    },
  };
}
```

```ts
export function createRuntimeServer(deps: RuntimeDeps = {}): Server {
  const gateway = deps.gateway ?? createGateway();
  const storage = deps.storage ?? createStorage();
  const tools = deps.tools ?? createTools(process.env, storage);
  // runtime 单实例持有一份确认闸门；连接 close 时按 runId 取消挂起。

  // 未设置时仅靠入口默认绑本机（见 isEntry）限制可达面。
  const authorized = (req: IncomingMessage): boolean => {
    const token = process.env.AGENT_RUNTIME_TOKEN;
    return !token || req.headers['x-agent-runtime-token'] === token;
  };

    if (req.method === 'GET' && req.url === '/health') {
      res.writeHead(200, { 'content-type': 'application/json' });
      res.end(JSON.stringify({ ok: true }));
      return;
    }

    if (req.method === 'POST' && req.url === '/runs') {
      if (!authorized(req)) {
        res.writeHead(401, { 'content-type': 'application/json' });
        res.end(JSON.stringify({ error: 'unauthorized' }));
        return;
      }
      let body = '';
      req.on('data', (chunk) => (body += chunk));
      req.on('end', () => {
        let runReq: RunRequest;
        try {
          runReq = JSON.parse(body || '{}') as RunRequest;
        } catch {
          res.writeHead(400, { 'content-type': 'application/json' });
          res.end(JSON.stringify({ error: 'invalid json' }));
          return;
        }

          'cache-control': 'no-cache',
          connection: 'keep-alive',

              gateway,
              tools,
              confirm,
              confirmTimeoutMs,
              onUsage: (u) => res.write(encodeSse({ event: 'usage', data: u })),
            })) {
              res.write(encodeSse({ event: 'step', data: ev }));
            }
            res.write(encodeSse({ event: 'done', data: { runId: runReq.runId } }));
          } catch (err) {
            res.write(encodeSse({ event: 'error', data: { message: String(err) } }));
          } finally {
            res.end();
          }
        })();
      });
      return;
    }

    if (confirmMatch) {
      if (!authorized(req)) {
        res.writeHead(401, { 'content-type': 'application/json' });
        res.end(JSON.stringify({ error: 'unauthorized' }));
        return;
      }
      const runId = decodeURIComponent(confirmMatch[1]);
      let body = '';
      req.on('data', (chunk) => (body += chunk));
      req.on('end', () => {
        let decision: { confirmId?: unknown; approved?: unknown };
        try {
          decision = JSON.parse(body || '{}') as typeof decision;
        } catch {
          res.writeHead(400, { 'content-type': 'application/json' });
          res.end(JSON.stringify({ error: 'invalid json' }));
          return;
        }
        if (
          typeof decision.confirmId !== 'string' ||
          decision.confirmId === '' ||
          typeof decision.approved !== 'boolean'
        ) {
          res.writeHead(400, { 'content-type': 'application/json' });
          res.end(JSON.stringify({ error: 'invalid_decision' }));
          return;
        }
        // runtime 无 run 注册表，区分不了「未知 run」与「已决」；二者统一 matched:false（幂等安全）。
        const matched = confirm.resolve(runId, decision.confirmId, decision.approved);
        res.writeHead(200, { 'content-type': 'application/json' });
        res.end(JSON.stringify({ ok: true, matched }));
      });
      return;
    }

    res.writeHead(404, { 'content-type': 'application/json' });
    res.end(JSON.stringify({ error: 'not found' }));
  });
```

```ts
import { Body, Controller, Param, Post, Res, UseGuards } from '@nestjs/common';
import type { FastifyReply } from 'fastify';
import type { ConfirmDecision } from '@ai-office/shared';
import { TasksService } from '../tasks/tasks.service';
import { JwtGuard } from '../auth/jwt.guard';
import { CurrentUser } from '../auth/current-user';
import type { JwtPayload } from '../auth/jwt';

@Controller('api/runs')
@UseGuards(JwtGuard)
export class RunsController {
  constructor(private readonly tasks: TasksService) {}

  @Post(':id/confirm')
  async confirm(
    @CurrentUser() user: JwtPayload,
    @Param('id') runId: string,
    @Body() body: ConfirmDecision,
    @Res() reply: FastifyReply,
  ): Promise<void> {
    const confirmId = body?.confirmId;
    const approved = body?.approved;
    if (typeof confirmId !== 'string' || confirmId === '' || typeof approved !== 'boolean') {
      reply.status(400).send({ error: 'invalid_decision' });
      return;
    }

    // 归属校验：非本人 run 回 404（不泄露存在性）。
    try {
      await this.tasks.assertRunOwned(runId, user.tenantId, user.sub);
    } catch {
      reply.status(404).send({ error: 'run_not_found' });
      return;
    }

    const runtimeUrl = process.env.AGENT_RUNTIME_URL ?? 'http://localhost:4001';
    const runtimeToken = process.env.AGENT_RUNTIME_TOKEN;
    let upstream: Response;
    try {
      upstream = await fetch(`${runtimeUrl}/runs/${encodeURIComponent(runId)}/confirm`, {
        method: 'POST',
        headers: {
          'content-type': 'application/json',
          ...(runtimeToken ? { 'x-agent-runtime-token': runtimeToken } : {}),
        },
        body: JSON.stringify({ confirmId, approved }),
      });
    } catch {
      reply.status(502).send({ error: 'upstream_unreachable' });
      return;
    }

    if (!upstream.ok) {
      // 上游非 2xx：原样透传状态码与响应体。
      const text = await upstream.text();
      reply
        .status(upstream.status)
        .header('content-type', upstream.headers.get('content-type') ?? 'application/json')
        .send(text || JSON.stringify({ error: 'upstream_error' }));
      return;
    }
    reply
      .status(200)
      .header('content-type', 'application/json')
      .send(await upstream.text());
  }
}
```

```ts
import type { AgentEvent } from '@ai-office/shared';

export interface ConfirmCard {
  confirmId: string;
  tool: string;
  args: Record<string, unknown>;
}

/**
 * 从事件流里找出**尚未收到 tool_result** 的确认请求。
 * 工具调用在单个 run 内串行，故 `confirm_request` 之后紧邻的下一个 `tool_result`
 * 必是它的决策结果——据此判定“是否仍在等待确认”，无需在 tool_result 里另带 confirmId。
 */
export function currentConfirmCard(events: AgentEvent[]): ConfirmCard | null {
  let card: ConfirmCard | null = null;
  for (const ev of events) {
    if (ev.type === 'confirm_request') {
      card = {
        confirmId: String(ev.payload.confirmId ?? ''),
        tool: String(ev.payload.tool ?? ''),
        args: (ev.payload.args as Record<string, unknown> | undefined) ?? {},
      };
    } else if (ev.type === 'tool_result') {
      card = null;
    }
  }
  return card;
}
```
