# M4b 连接器 / MCP 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让 Agent 能接入外部 MCP Server —— 新增连接器 → 自动发现其工具 → 并入 runtime 注册表 → 模型可调用（验收：接入一个 MCP Server 后其工具可用）。

**Architecture:** 维持 M1–M4a 的三层解耦与「BFF 独占 DB、runtime 只执行」边界。连接器配置（含 AES-256-GCM 加密的 `env`）存 BFF 的 `Connector` 表；runtime **每次 run 开始**从 BFF 内部端点 `GET /internal/connectors` 拉该租户启用连接器（解密配置），由 `McpHost` 托管常驻 MCP 子进程（stdio + 心跳 + 崩溃退避重拉），把发现的工具以 `mcp.<server>.<tool>`（`permission:'confirm'`）并入本次 run 的工具列表。新增/编辑连接器时 BFF 同步调 runtime `POST /connectors/validate`（临时子进程 spawn + `initialize` + `tools/list`）校验，成功才落库。

**Tech Stack:** TypeScript 5.6 (ESM), Node ≥22, NestJS 10 + Fastify 4, Prisma 6 + PostgreSQL 16, Next.js 15 / React 19, vitest 2, pnpm + Turborepo, `@modelcontextprotocol/sdk` ^1.32.1。

**关联文档:** `docs/superpowers/specs/2026-10-16-m4b-mcp-design.md`（本计划即其实施）。

**关键约束（务必遵守）**
- **直接在 master 开发**，不用特性分支 / worktree。
- **绝不 `git add -A` / `git add .`**：仓库根有未跟踪的 `任务.txt`，绝不能提交；一律显式指定路径。
- commit 信息用中文：`type(scope): 摘要`，正文写「为什么」。
- 所有文件操作使用**完整 Windows 绝对路径**。
- **只有 `@modelcontextprotocol/sdk` 是新增依赖**，加在 `apps/agent-runtime`。
- BFF 的 NestJS **构造函数参数若是 TS `interface`/类型别名**，`design:paramtypes` 会退化成 `Object` 导致 DI 解析失败——`connector.service` 的第二参（`ConnectorRuntimeConfig`）必须像 `kb.service` 那样用 `@Optional()` 装饰，且测试与生产都用**两参构造**。
- runtime 的 `env` 传给 MCP 子进程时**不得注入完整 `process.env`**：以 SDK `getDefaultEnvironment()`（安全白名单）为基底再合并连接器自身 `env`，避免泄露 `DEEPSEEK_API_KEY`/`JWT_SECRET`。
- `Prisma` 的 `Json` 字段读回来是 `Prisma.JsonValue`，映射 `string[]` 时必须 `Array.isArray(x) ? x as string[] : []`。
- 三处测试跑法：`pnpm --filter @ai-office/bff test`、`pnpm --filter @ai-office/agent-runtime test`、`pnpm --filter @ai-office/web test`。

- [ ] **Step 3: 实现 `crypto.ts`**

```ts
import { createCipheriv, createDecipheriv, randomBytes } from 'node:crypto';

const ALGO = 'aes-256-gcm';
const IV_LEN = 12;
const TAG_LEN = 16;

/** 读密钥并校验；缺或非 32 字节 hex 即抛错（fail-fast）。 */
function key(): Buffer {
  const hex = process.env.CONNECTOR_ENC_KEY;
  if (!hex) throw new Error('CONNECTOR_ENC_KEY 未设置');
  const buf = Buffer.from(hex, 'hex');
  if (buf.length !== 32) throw new Error('CONNECTOR_ENC_KEY 必须是 32 字节 hex');
  return buf;
}

/** 启动期自检：缺/非法密钥直接崩，避免运行到首次写连接器才暴露。 */
export function assertConnectorKey(): void {
  key();
}

/** 加密为 base64(IV ‖ authTag ‖ ciphertext)。 */
export function encrypt(plain: string): string {
  const iv = randomBytes(IV_LEN);
  const cipher = createCipheriv(ALGO, key(), iv);
  const ciphertext = Buffer.concat([cipher.update(plain, 'utf8'), cipher.final()]);
  const tag = cipher.getAuthTag();
  return Buffer.concat([iv, tag, ciphertext]).toString('base64');
}

/** 解密 base64(IV ‖ authTag ‖ ciphertext)；篡改/换密钥时抛错。 */
export function decrypt(blob: string): string {
  const buf = Buffer.from(blob, 'base64');
  const iv = buf.subarray(0, IV_LEN);
  const tag = buf.subarray(IV_LEN, IV_LEN + TAG_LEN);
  const ciphertext = buf.subarray(IV_LEN + TAG_LEN);
  const decipher = createDecipheriv(ALGO, key(), iv);
  decipher.setAuthTag(tag);
  return Buffer.concat([decipher.update(ciphertext), decipher.final()]).toString('utf8');
}
```

```ts
import { CanActivate, ExecutionContext, ForbiddenException, Injectable } from '@nestjs/common';
import { PrismaService } from '../db/prisma.service';
import type { AuthedRequest } from './jwt.guard';

/**
 * 管理员校验：由 JWT `sub` 查库取 `role`（不入 JWT，改角色无需重发 token）。
 * **须置于 JwtGuard 之后**（`@UseGuards(JwtGuard, AdminGuard)`），依赖其已填充 req.user。
 */
@Injectable()
export class AdminGuard implements CanActivate {
  constructor(private readonly prisma: PrismaService) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const req = context.switchToHttp().getRequest<Partial<AuthedRequest>>();
    const sub = req.user?.sub;
    if (!sub) throw new ForbiddenException('not_admin');
    const user = await this.prisma.user.findUnique({ where: { id: sub }, select: { role: true } });
    if (user?.role !== 'admin') throw new ForbiddenException('not_admin');
    return true;
  }
}
```

```ts
import { ConflictException, Injectable, Logger, NotFoundException, Optional } from '@nestjs/common';
import type { ConnectorDto, McpToolInfo } from '@ai-office/shared';
import { PrismaService } from '../db/prisma.service';
import { encrypt, decrypt } from './crypto';

const VALIDATE_TIMEOUT_MS = 20_000;

export interface ConnectorRuntimeConfig {
  runtimeUrl?: string;
  runtimeToken?: string;
  fetchImpl?: typeof fetch;
}

export interface ConnectorInput {
  name: string;
  command: string;
  args?: string[];
  env?: Record<string, string>;
}

export interface ConnectorPatch {
  name?: string;
  command?: string;
  args?: string[];
  env?: Record<string, string>;
  enabled?: boolean;
}

/** runtime 视角的完整连接器（含解密 env）。 */
export interface EnabledConnector {
  id: string;
  name: string;
  command: string;
  args: string[];
  env: Record<string, string>;
  enabled: boolean;
}

/** runtime 校验失败（spawn/握手/协议错）：由调用方映射为 400。 */
export class ValidateFailedError extends Error {}

@Injectable()
export class ConnectorService {
  private readonly logger = new Logger(ConnectorService.name);

  constructor(
    private readonly prisma: PrismaService,
    @Optional() private readonly cfg: ConnectorRuntimeConfig = {},
  ) {}

  async create(input: ConnectorInput, tenantId: string, createdBy: string): Promise<ConnectorDto> {
    const dup = await this.prisma.connector.findFirst({ where: { tenantId, name: input.name } });
    if (dup) throw new ConflictException('duplicate_name');
    const args = input.args ?? [];
    const env = input.env ?? {};
    const tools = await this.validateRuntime(input.command, args, env);
    try {
      const row = await this.prisma.connector.create({
        data: {
          tenantId,
          name: input.name,
          command: input.command,
          args,
          envEncrypted: encrypt(JSON.stringify(env)),
          enabled: true,
          toolCount: tools.length,
          lastValidatedAt: new Date(),
          createdBy,
        },
      });
      return toDto(row);
    } catch (e) {
      // 并发下 findFirst 去重可能被窗口绕过，唯一约束兜底为 409 而非 500。
      if (isUniqueViolation(e)) throw new ConflictException('duplicate_name');
      throw e;
    }
  }

  async list(tenantId: string): Promise<ConnectorDto[]> {
    const rows = await this.prisma.connector.findMany({
      where: { tenantId },
      orderBy: { createdAt: 'desc' },
    });
    return rows.map(toDto);
  }

  async update(id: string, tenantId: string, patch: ConnectorPatch): Promise<ConnectorDto> {
    const row = await this.prisma.connector.findFirst({ where: { id, tenantId } });
    if (!row) throw new NotFoundException('connector_not_found');

    // 改名撞重名须显式拦截：否则触及 @@unique([tenantId,name]) 会抛裸 Prisma 错、映射成 500。
    if (patch.name !== undefined && patch.name !== row.name) {
      const dup = await this.prisma.connector.findFirst({ where: { tenantId, name: patch.name } });
      if (dup) throw new ConflictException('duplicate_name');
    }

    const data: Record<string, unknown> = {};
    if (patch.name !== undefined) data.name = patch.name;
    if (patch.enabled !== undefined) data.enabled = patch.enabled;

    // 仅当「会改变运行行为」的字段变化时才重新校验（切 enabled 不必）。
    const touchesRuntime =
      patch.command !== undefined || patch.args !== undefined || patch.env !== undefined;
    if (touchesRuntime) {
      const command = patch.command ?? row.command;
      const args = patch.args ?? readStringArray(row.args);
      const env =
        patch.env ?? (JSON.parse(decrypt(row.envEncrypted)) as Record<string, string>);
      const tools = await this.validateRuntime(command, args, env);
      data.command = command;
      data.args = args;
      if (patch.env !== undefined) data.envEncrypted = encrypt(JSON.stringify(env));
      data.toolCount = tools.length;
      data.lastValidatedAt = new Date();
    }

    try {
      const updated = await this.prisma.connector.update({ where: { id }, data });
      return toDto(updated);
    } catch (e) {
      if (isUniqueViolation(e)) throw new ConflictException('duplicate_name');
      throw e;
    }
  }

  async remove(id: string, tenantId: string): Promise<void> {
    const row = await this.prisma.connector.findFirst({ where: { id, tenantId }, select: { id: true } });
    if (!row) throw new NotFoundException('connector_not_found');
    await this.prisma.connector.delete({ where: { id } });
  }

  /** 供 runtime 拉取：启用连接器，env 解密。 */
  async listEnabled(tenantId: string): Promise<EnabledConnector[]> {
    const rows = await this.prisma.connector.findMany({ where: { tenantId, enabled: true } });
    return rows.map((r) => ({
      id: r.id,
      name: r.name,
      command: r.command,
      args: readStringArray(r.args),
      env: JSON.parse(decrypt(r.envEncrypted)) as Record<string, string>,
      enabled: r.enabled,
    }));
  }

  private async validateRuntime(
    command: string,
    args: string[],
    env: Record<string, string>,
  ): Promise<McpToolInfo[]> {
    const doFetch = this.cfg.fetchImpl ?? fetch;
    const base = this.cfg.runtimeUrl ?? process.env.AGENT_RUNTIME_URL ?? 'http://localhost:4001';
    const token = this.cfg.runtimeToken ?? process.env.AGENT_RUNTIME_TOKEN;
    let res: Response;
    try {
      res = await doFetch(`${base}/connectors/validate`, {
        method: 'POST',
        headers: {
          'content-type': 'application/json',
          ...(token ? { 'x-agent-runtime-token': token } : {}),
        },
        body: JSON.stringify({ command, args, env }),
        signal: AbortSignal.timeout(VALIDATE_TIMEOUT_MS),
      });
    } catch {
      throw new ValidateFailedError('runtime_unreachable');
    }
    if (!res.ok) {
      await res.body?.cancel().catch(() => undefined);
      throw new ValidateFailedError(`runtime_http_${res.status}`);
    }
    let data: { ok: boolean; tools?: McpToolInfo[]; error?: string };
    try {
      data = (await res.json()) as { ok: boolean; tools?: McpToolInfo[]; error?: string };
    } catch {
      throw new ValidateFailedError('invalid_response');
    }
    if (!data.ok) {
      // 不透传 runtime 原始错误：其内容可能回显 command/env，构成密钥泄露面，仅记服务端日志。
      this.logger.warn(`连接器校验失败：${data.error ?? 'unknown'}`);
      throw new ValidateFailedError('validate_failed');
    }
    return data.tools ?? [];
  }
}

function isUniqueViolation(e: unknown): boolean {
  return typeof e === 'object' && e !== null && (e as { code?: unknown }).code === 'P2002';
}

function readStringArray(v: unknown): string[] {
  return Array.isArray(v) ? v.filter((x): x is string => typeof x === 'string') : [];
}

function toDto(row: {
  id: string;
  name: string;
  command: string;
  args: unknown;
  enabled: boolean;
  toolCount: number;
  lastValidatedAt: Date | null;
  createdAt: Date;
}): ConnectorDto {
  return {
    id: row.id,
    name: row.name,
    command: row.command,
    args: readStringArray(row.args),
    enabled: row.enabled,
    toolCount: row.toolCount,
    lastValidatedAt: row.lastValidatedAt ? row.lastValidatedAt.toISOString() : null,
    createdAt: row.createdAt.toISOString(),
  };
}
```

```ts
import {
  Body,
  ConflictException,
  Controller,
  Delete,
  Get,
  HttpCode,
  NotFoundException,
  Param,
  Patch,
  Post,
  Res,
  UseGuards,

import type { FastifyReply } from 'fastify';
import type { ConnectorDto } from '@ai-office/shared';
import { JwtGuard } from '../auth/jwt.guard';
import { AdminGuard } from '../auth/admin.guard';
import { CurrentUser } from '../auth/current-user';
import type { JwtPayload } from '../auth/jwt';
import { ConnectorService, ValidateFailedError, type ConnectorInput, type ConnectorPatch } from './connector.service';

@Controller('api/connectors')
@UseGuards(JwtGuard, AdminGuard)
export class ConnectorController {
  constructor(private readonly connectors: ConnectorService) {}

  @Get()
  list(@CurrentUser() user: JwtPayload): Promise<ConnectorDto[]> {
    return this.connectors.list(user.tenantId);
  }

  @Post()
  async create(
    @CurrentUser() user: JwtPayload,
    @Body() body: ConnectorInput,
    @Res() reply: FastifyReply,
  ): Promise<void> {
    try {
      const dto = await this.connectors.create(body, user.tenantId, user.sub);
      reply.status(201).send(dto);
    } catch (err) {
      sendMapped(err, reply);
    }
  }

  @Patch(':id')
  async update(
    @CurrentUser() user: JwtPayload,
    @Param('id') id: string,
    @Body() body: ConnectorPatch,
    @Res() reply: FastifyReply,
  ): Promise<void> {
    try {
      reply.status(200).send(await this.connectors.update(id, user.tenantId, body));
    } catch (err) {
      sendMapped(err, reply, 'connector_not_found');
    }
  }

  @Delete(':id')
  @HttpCode(204)
  async remove(
    @CurrentUser() user: JwtPayload,
    @Param('id') id: string,
    @Res() reply: FastifyReply,
  ): Promise<void> {
    try {
      await this.connectors.remove(id, user.tenantId);
      reply.status(204).send();
    } catch (err) {
      sendMapped(err, reply, 'connector_not_found');
    }
  }
}

/** 把 service 抛出的领域异常映射为 HTTP 响应；未识别的一律重抛由 Nest 兜底。 */
function sendMapped(err: unknown, reply: FastifyReply, notFound = 'connector_not_found'): void {
  if (err instanceof ValidateFailedError) {
    // ValidateFailedError.message 已是固定码（Task 5 起不再回传 runtime 原始错误），无需再透传 reason。
    reply.status(400).send({ error: 'validate_failed' });
    return;
  }
  if (err instanceof ConflictException) {
    reply.status(400).send({ error: 'duplicate_name' });
    return;
  }
  if (err instanceof NotFoundException) {
    reply.status(404).send({ error: notFound });
    return;
  }
  throw err;
}
```

```ts
import type { FastifyRequest, FastifyReply } from 'fastify';
import { ConnectorService, type EnabledConnector } from './connector.service';
import { isLoopback } from '../kb/loopback';

/**
 * 内部端点：供 runtime 拉取启用连接器（含解密 env）。
 * 与 M4a 一致，fail-closed：配了 INTERNAL_TOKEN 用共享密钥；未配则仅放行回环来源。
 */
@Controller('internal/connectors')
export class InternalConnectorController {
  constructor(private readonly connectors: ConnectorService) {}

  @Get()
  async list(
    @Req() req: FastifyRequest,
    @Query('tenantId') tenantId: string | undefined,
    @Res() reply: FastifyReply,
  ): Promise<void> {
    const token = process.env.INTERNAL_TOKEN;
    if (token) {
      if (req.headers['x-internal-token'] !== token) {
        reply.status(401).send({ error: 'unauthorized' });
        return;
      }
    } else if (!isLoopback(req.ip)) {
      reply.status(403).send({ error: 'forbidden' });
      return;
    }
    if (!tenantId) {
      reply.status(400).send({ error: 'missing_tenant' });
      return;
    }
    const list: EnabledConnector[] = await this.connectors.listEnabled(tenantId);
    reply.status(200).send(list);
  }
}
```

```ts
import { Client } from '@modelcontextprotocol/sdk/client/index.js';
import {
  StdioClientTransport,
  getDefaultEnvironment,
} from '@modelcontextprotocol/sdk/client/stdio.js';
import type { McpToolInfo } from '@ai-office/shared';
import type { ConnectorConfig, McpTransport } from './transport';

export interface StdioTransportOptions {
  initTimeoutMs: number;
  callTimeoutMs: number;
}

/**
 * 子进程环境 = SDK 白名单（PATH/HOME 等）合并连接器 env。
 * **绝不注入完整 process.env**：否则会把 DEEPSEEK_API_KEY / JWT_SECRET 等
 * 本服务密钥泄露给任意第三方 MCP Server。
 */
export function buildChildEnv(connectorEnv: Record<string, string>): Record<string, string> {
  return { ...getDefaultEnvironment(), ...connectorEnv };
}

/** 带超时的 promise 包装；超时抛 `<label>_timeout` 且不再等待原 promise。 */
function withTimeout<T>(p: Promise<T>, ms: number, label: string): Promise<T> {
  return new Promise<T>((resolve, reject) => {
    const t = setTimeout(() => reject(new Error(`${label}_timeout`)), ms);
    t.unref?.();
    p.then(
      (v) => { clearTimeout(t); resolve(v); },
      (e) => { clearTimeout(t); reject(e); },
    );
  });
}

/** stdio MCP 传输：封装官方 SDK Client + StdioClientTransport。 */
export class StdioMcpTransport implements McpTransport {
  private readonly client: Client;

  constructor(
    private readonly cfg: ConnectorConfig,
    private readonly opts: StdioTransportOptions,
  ) {
    this.client = new Client({ name: 'ai-office-agent', version: '1.0.0' });
  }

  async connect(): Promise<void> {
    const transport = new StdioClientTransport({
      command: this.cfg.command,
      args: this.cfg.args,
      env: buildChildEnv(this.cfg.env),
    });
    await withTimeout(this.client.connect(transport), this.opts.initTimeoutMs, 'mcp_init');
  }

  async listTools(): Promise<McpToolInfo[]> {
    const { tools } = await this.client.listTools();
    return tools.map((t) => ({
      name: t.name,
      description: t.description ?? '',
      inputSchema: (t.inputSchema as Record<string, unknown>) ?? { type: 'object', properties: {} },
    }));
  }

  async callTool(name: string, args: Record<string, unknown>): Promise<{ text: string; isError: boolean }> {
    const res = await withTimeout(
      this.client.callTool({ name, arguments: args }),
      this.opts.callTimeoutMs,
      'mcp_call',
    );
    const content = (res.content ?? []) as { type?: string; text?: string }[];
    const text = content
      .filter((c) => c.type === 'text')
      .map((c) => c.text ?? '')
      .join('\n');
    return { text, isError: res.isError === true };
  }

  async ping(): Promise<void> {
    await this.client.ping();
  }

  async close(): Promise<void> {
    await this.client.close();
  }
}
```

```ts
import type { McpToolInfo } from '@ai-office/shared';
import type { ConnectorConfig, McpTransport, TransportFactory } from './transport';
import { StdioMcpTransport } from './stdio-transport';

export interface McpHostOptions {
  heartbeatMs: number;
  backoffBaseMs: number;
  backoffMaxMs: number;
  initTimeoutMs: number;
  callTimeoutMs: number;
  /** 测试注入；默认建 stdio 传输。 */
  transportFactory?: TransportFactory;
}

type State = 'starting' | 'ready' | 'error';

interface Entry {
  cfg: ConnectorConfig;
  transport: McpTransport | null;
  state: State;
  tools: McpToolInfo[];
  backoffMs: number;
  heartbeatTimer?: ReturnType<typeof setInterval>;
  retryTimer?: ReturnType<typeof setTimeout>;
}

/**
 * MCP 子进程守护池：按 serverName 复用常驻子进程，心跳巡检，
 * 崩溃后指数退避重拉。实例随 runtime 进程存活（见 server.ts closeAll）。
 */
export class McpHost {
  private readonly entries = new Map<string, Entry>();
  private closed = false;

  constructor(private readonly opts: McpHostOptions) {}

  private factory(cfg: ConnectorConfig): McpTransport {
    if (this.opts.transportFactory) return this.opts.transportFactory(cfg);
    return new StdioMcpTransport(cfg, {
      initTimeoutMs: this.opts.initTimeoutMs,
      callTimeoutMs: this.opts.callTimeoutMs,
    });
  }

  /** 确保某连接器子进程就绪并返回其工具；失败抛错并安排退避重拉。 */
  async ensure(cfg: ConnectorConfig): Promise<McpToolInfo[]> {
    const existing = this.entries.get(cfg.name);
    if (existing?.state === 'ready') return existing.tools;

    const transport = this.factory(cfg);
    const prevBackoff = existing?.backoffMs ?? this.opts.backoffBaseMs;
    this.entries.set(cfg.name, {
      cfg, transport, state: 'starting', tools: [], backoffMs: prevBackoff,
    });
    try {
      await transport.connect();
      const tools = await transport.listTools();
      const entry = this.entries.get(cfg.name)!;
      entry.state = 'ready';
      entry.tools = tools;
      entry.backoffMs = this.opts.backoffBaseMs; // 成功后重置退避
      this.startHeartbeat(cfg.name, entry);
      return tools;
    } catch (err) {
      await transport.close().catch(() => undefined);
      this.markError(cfg.name);
      throw err;
    }
  }

  async call(
    server: string,
    tool: string,
    args: Record<string, unknown>,
  ): Promise<{ text: string; isError: boolean }> {
    const entry = this.entries.get(server);
    if (!entry || entry.state !== 'ready' || !entry.transport) {
      throw new Error('mcp_server_unavailable');
    }
    return entry.transport.callTool(tool, args);
  }

  async closeAll(): Promise<void> {
    this.closed = true;
    for (const entry of this.entries.values()) {
      if (entry.heartbeatTimer) clearInterval(entry.heartbeatTimer);
      if (entry.retryTimer) clearTimeout(entry.retryTimer);
      await entry.transport?.close().catch(() => undefined);
    }
    this.entries.clear();
  }

  private startHeartbeat(name: string, entry: Entry): void {
    entry.heartbeatTimer = setInterval(() => void this.beat(name), this.opts.heartbeatMs);
    entry.heartbeatTimer.unref?.();
  }

  private async beat(name: string): Promise<void> {
    const entry = this.entries.get(name);
    if (!entry || entry.state !== 'ready' || !entry.transport) return;
    try {
      await entry.transport.ping();
    } catch {
      if (entry.heartbeatTimer) clearInterval(entry.heartbeatTimer);
      await entry.transport.close().catch(() => undefined);
      entry.transport = null;
      this.markError(name);
    }
  }

  /** 置 error 并安排一次指数退避重拉；已 closed 则不再排期。 */
  private markError(name: string): void {
    if (this.closed) return;
    const entry = this.entries.get(name);
    if (!entry) return;
    if (entry.heartbeatTimer) clearInterval(entry.heartbeatTimer);
    entry.state = 'error';
    entry.tools = [];
    const delay = Math.min(entry.backoffMs, this.opts.backoffMaxMs);
    entry.backoffMs = Math.min(entry.backoffMs * 2, this.opts.backoffMaxMs);
    const cfg = entry.cfg;
    entry.retryTimer = setTimeout(() => void this.ensure(cfg).catch(() => undefined), delay);
    entry.retryTimer.unref?.();
  }
}
```

```ts
export interface BffConnectorClientConfig {
  baseUrl: string;

/** 调 BFF 内部端点拉取某租户启用的连接器（env 已解密）；非 2xx 一律抛错。 */
export async function fetchConnectors(
  cfg: BffConnectorClientConfig,
```

```ts
import type { Tool } from '../tools/types';
import type { McpHost } from './host';
import type { ConnectorConfig } from './transport';

/**
 * 把连接器聚合为本次 run 的 MCP 工具：名 `mcp.<server>.<tool>`、`permission:'confirm'`。
 * 单个 Server 拉起失败只跳过它，不影响其他工具与 agent loop。
 */
export async function buildMcpTools(
  connectors: ConnectorConfig[],
  host: McpHost,
): Promise<Tool[]> {
  const tools: Tool[] = [];
  for (const cfg of connectors) {
    if (!cfg.enabled) continue;
    let infos;
    try {
      infos = await host.ensure(cfg);
    } catch {
      continue; // 记日志由 host 承担；此处静默跳过该 Server。
    }
    for (const info of infos) {
      const name = `mcp.${cfg.name}.${info.name}`;
      tools.push({
        name,
        description: info.description || `MCP 工具 ${cfg.name}/${info.name}`,
        parameters: info.inputSchema ?? { type: 'object', properties: {} },
        permission: 'confirm',
        async execute(args): Promise<unknown> {
          try {
            const out = await host.call(cfg.name, info.name, args);
            return { ok: !out.isError, tool: name, text: out.text };
          } catch {
            // 调用失败不中断 loop：回结构化失败，模型据此说明。
            return { ok: false, tool: name, error: 'mcp_call_failed', text: '外部工具暂时不可用' };
          }
        },
      });
    }
  }
  return tools;
}
```

```ts
  args: string[];

/**

export async function validateConnector(
  input: ValidateInput,
  opts: { initTimeoutMs: number },

  });

    await transport.connect();
    return await transport.listTools();
```

```tsx
'use client';

import { useCallback, useEffect, useState } from 'react';
import { useRouter } from 'next/navigation';
import type { ConnectorDto } from '@ai-office/shared';
import {
  createConnector,
  deleteConnector,
  getRole,
  getToken,
  isUnauthorized,
  listConnectors,
  updateConnector,
} from '../../lib/api';

type EnvRow = { key: string; value: string };

export default function ConnectorsPage() {
  const router = useRouter();
  const [items, setItems] = useState<ConnectorDto[]>([]);
  const [error, setError] = useState('');
  const [busy, setBusy] = useState(false);
  const [name, setName] = useState('');
  const [command, setCommand] = useState('');
  const [argsText, setArgsText] = useState('');
  const [envRows, setEnvRows] = useState<EnvRow[]>([{ key: '', value: '' }]);

  const isAdmin = getRole() === 'admin';

  const refresh = useCallback(async () => {
    try {
      setItems(await listConnectors());
    } catch (err) {
      if (isUnauthorized(err)) {
        router.push('/login');
        return;
      }
      setError(String(err));
    }
  }, [router]);

  useEffect(() => {
    if (!getToken()) {
      router.push('/login');
      return;
    }
    void refresh();
  }, [refresh, router]);

  async function submit() {
    setBusy(true);
    setError('');
    try {
      const env = Object.fromEntries(
        envRows.filter((r) => r.key.trim()).map((r) => [r.key.trim(), r.value]),
      );
      const args = argsText.trim() ? argsText.trim().split(/\s+/) : [];
      const created = await createConnector({ name: name.trim(), command: command.trim(), args, env });
      setItems((prev) => [created, ...prev]);
      setName('');
      setCommand('');
      setArgsText('');
      setEnvRows([{ key: '', value: '' }]);
    } catch (err) {
      if (isUnauthorized(err)) {
        router.push('/login');
        return;
      }
      setError(String(err));
    } finally {
      setBusy(false);
    }
  }

  async function toggle(item: ConnectorDto) {
    try {
      const updated = await updateConnector(item.id, { enabled: !item.enabled });
      setItems((prev) => prev.map((c) => (c.id === updated.id ? updated : c)));
    } catch (err) {
      if (!isUnauthorized(err)) setError(String(err));
    }
  }

  async function remove(id: string) {
    try {
      await deleteConnector(id);
      setItems((prev) => prev.filter((c) => c.id !== id));
    } catch (err) {
      if (isUnauthorized(err)) {
        router.push('/login');
        return;
      }
      setError(String(err));
    }
  }

  return (
    <main>
      <h1>连接器</h1>
      <button onClick={() => router.push('/')} style={{ marginBottom: 16 }}>
        返回工作台
      </button>
      {error && <p style={{ color: 'crimson' }}>{error}</p>}

      {isAdmin ? (
        <section style={{ border: '1px solid #ddd', borderRadius: 6, padding: 12, marginBottom: 16 }}>
          <h2 style={{ fontSize: 16 }}>新增 MCP Server</h2>
          <div style={{ display: 'flex', flexDirection: 'column', gap: 8, maxWidth: 480 }}>
            <input placeholder="名称（租户内唯一，如 filesystem）" value={name} onChange={(e) => setName(e.target.value)} />
            <input placeholder="命令（如 npx）" value={command} onChange={(e) => setCommand(e.target.value)} />
            <input placeholder="参数（空格分隔，如 -y @modelcontextprotocol/server-filesystem /data）" value={argsText} onChange={(e) => setArgsText(e.target.value)} />
            <div>
              <div style={{ fontSize: 13, color: '#666' }}>环境变量（密钥）</div>
              {envRows.map((row, i) => (
                <div key={i} style={{ display: 'flex', gap: 8, marginTop: 4 }}>
                  <input placeholder="KEY" value={row.key} onChange={(e) => setEnvRows((p) => p.map((r, j) => (j === i ? { ...r, key: e.target.value } : r)))} />
                  <input placeholder="VALUE" type="password" value={row.value} onChange={(e) => setEnvRows((p) => p.map((r, j) => (j === i ? { ...r, value: e.target.value } : r)))} />
                  <button onClick={() => setEnvRows((p) => p.filter((_, j) => j !== i))}>删除</button>
                </div>
              ))}
              <button style={{ marginTop: 4 }} onClick={() => setEnvRows((p) => [...p, { key: '', value: '' }])}>+ 变量</button>
            </div>
            <button onClick={() => void submit()} disabled={busy || !name.trim() || !command.trim()}>
              {busy ? '校验中…' : '新增'}
            </button>
          </div>
        </section>
      ) : (
        <p style={{ color: '#888' }}>仅管理员可管理连接器，当前为只读。</p>
      )}

      {items.length === 0 ? (
        <p style={{ color: '#888' }}>暂无连接器</p>
      ) : (
        <ul style={{ listStyle: 'none', padding: 0 }}>
          {items.map((c) => (
            <li key={c.id} style={{ border: '1px solid #ddd', borderRadius: 6, padding: 12, marginBottom: 8 }}>
              <div style={{ fontWeight: 600 }}>{c.name}</div>
              <div style={{ color: '#666', fontSize: 13 }}>
                {c.command} {c.args.join(' ')} · {c.toolCount} 个工具
                {c.lastValidatedAt ? ` · 校验于 ${new Date(c.lastValidatedAt).toLocaleString()}` : ''}
              </div>
              {isAdmin && (
                <div style={{ display: 'flex', gap: 8, marginTop: 8 }}>
                  <button onClick={() => void toggle(c)}>{c.enabled ? '停用' : '启用'}</button>
                  <button onClick={() => void remove(c.id)}>删除</button>
                </div>
              )}
            </li>
          ))}
        </ul>
      )}
    </main>
  );
}
```
