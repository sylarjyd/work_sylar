# M4a 知识库 RAG 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让 Agent 能基于用户私有文档作答——上传 → 解析/切分/向量化 → 落 pgvector → `kb.retrieve` 检索并带来源引用。

**Architecture:** 维持 M1–M3 的三层解耦与「BFF 独占 PG、runtime 无状态」边界。runtime 负责 CPU 密集的**解析与嵌入**（新增 `EMBEDDING_*` 网关，与 `ModelGateway` 同构），BFF 独占 **DB 与向量检索**（pgvector / HNSW，`KbDoc` + `KbChunk`）。上传异步：BFF 存 S3 建 `pending` 文档即回 202，后台调 runtime `/kb/ingest` 取字节→解析→切分→嵌入，再把切片（含向量）写库并置 `ready`。

**Tech Stack:** TypeScript 5.6 (ESM), Node ≥22, NestJS 10 + Fastify 4, Prisma 6 + PostgreSQL 16 + pgvector, Next.js 15 / React 19, vitest 2, pnpm + Turborepo。

**关键约束（务必遵守）**
- **直接在 master 开发**，不用特性分支 / worktree。
- **绝不 `git add -A` / `git add .`**：仓库根有未跟踪的 `任务.txt`，绝不能提交；一律显式指定路径。
- commit 信息用中文：`type(scope): 摘要`，正文写「为什么」。
- 所有文件操作使用**完整 Windows 绝对路径**。
- `KbChunk.vector` 是 Prisma `Unsupported("vector(1024)")`：**含必填 Unsupported 字段的模型不能用 `create()`**，切片写入与向量读取一律走 `$executeRaw`/`$queryRaw`。
- `RunRequest` / `RunContext` 需新增 `userId`（检索按 `tenant_id + user_id` 过滤；M3 只透传了 `tenantId`）。
- agent-loop 的 `preview()` 会把整个工具结果在 500 字处截断；因此工具结果里的 `display` 字段需在 agent-loop **原样透传**（不截断），供 web 渲染来源列表。

```ts
import type { EmbeddingGateway } from './types';

/** 词袋哈希：非空文本返回 1024 维单位向量，使词面重叠驱动余弦相似度（离线可验证管线）。 */
export function hashEmbed(text: string, dim = 1024): number[] {
  const v = new Array<number>(dim).fill(0);
  for (const tok of tokenize(text)) {
    // 双哈希降碰撞：同一词落两处，仍由词面重叠主导相似度。
    v[hash(tok, 0x811c9dc5) % dim] += 1;
    v[hash(tok, 0x1000193) % dim] += 1;
  }
  let norm = 0;
  for (const x of v) norm += x * x;
  norm = Math.sqrt(norm);
  return norm === 0 ? v : v.map((x) => x / norm);
}

/** FNV-1a 变体，带种子。 */
function hash(s: string, seed: number): number {
  let h = seed >>> 0;
  for (let i = 0; i < s.length; i++) {
    h ^= s.charCodeAt(i);
    h = Math.imul(h, 0x01000193) >>> 0;
  }
  return h;
}

/** 按 Unicode 字母/数字切词，其余为分隔符；全小写。 */
function tokenize(text: string): string[] {
  return text
    .toLowerCase()
    .split(/[^\p{L}\p{N}]+/u)
    .filter((t) => t.length > 0);
}

/** 两向量余弦（任一为零向量则返回 0）。 */
export function cosine(a: number[], b: number[]): number {
  let dot = 0;
  let na = 0;
  let nb = 0;
  for (let i = 0; i < a.length; i++) {
    dot += a[i] * b[i];
    na += a[i] * a[i];
    nb += b[i] * b[i];
  }
  if (na === 0 || nb === 0) return 0;
  return dot / (Math.sqrt(na) * Math.sqrt(nb));
}

/** 无 key / 测试用的确定性嵌入。 */
export class MockEmbeddingGateway implements EmbeddingGateway {
  readonly dim: number;
  constructor(dim = 1024) {
    this.dim = dim;
  }
  async embed(texts: string[]): Promise<number[][]> {
    return texts.map((t) => hashEmbed(t, this.dim));
  }
}
```

```ts
import type { EmbeddingGateway } from './types';

interface EmbeddingsResponse {
  data?: { index?: number; embedding: number[] }[];
}

export interface OpenAICompatEmbeddingConfig {
  baseUrl: string;
  apiKey: string;
  model: string;
  dim: number;
}

/** 走 OpenAI 兼容的 /embeddings（SiliconFlow / OpenAI / 兼容网关）。 */
export class OpenAICompatEmbeddingGateway implements EmbeddingGateway {
  readonly dim: number;
  constructor(private readonly cfg: OpenAICompatEmbeddingConfig) {
    this.dim = cfg.dim;
  }

  async embed(texts: string[]): Promise<number[][]> {
    const res = await fetch(`${this.cfg.baseUrl}/embeddings`, {
      method: 'POST',
      headers: {
        'content-type': 'application/json',
        authorization: `Bearer ${this.cfg.apiKey}`,
      },
      body: JSON.stringify({ model: this.cfg.model, input: texts }),
    });
    if (!res.ok) throw new Error(`embedding_http_${res.status}`);
    const data = (await res.json()) as EmbeddingsResponse;
    const rows = [...(data.data ?? [])];
    // provider 不保证顺序：按 index 升序还原，缺 index 时按原序。
    rows.sort((a, b) => (a.index ?? 0) - (b.index ?? 0));
    return rows.map((r) => r.embedding);
  }
}
```

```ts
import ExcelJS from 'exceljs';
import mammoth from 'mammoth';

export type DocType = 'txt' | 'md' | 'csv' | 'docx' | 'xlsx' | 'pdf';

export interface ParseInput {
  data: Buffer;
  filename: string;
  mime?: string;
}

/** pdf 抽取器可注入：单测用替身，生产走 pdfjs（避免携带真实 PDF 二进制 fixture）。 */
export interface ParseDeps {
  pdf?: (data: Buffer) => Promise<string>;
}

const EXT: Record<string, DocType> = {
  txt: 'txt',
  md: 'md',
  csv: 'csv',
  docx: 'docx',
  xlsx: 'xlsx',
  pdf: 'pdf',
};

const MIME: Record<string, DocType> = {
  'text/plain': 'txt',
  'text/markdown': 'md',
  'text/csv': 'csv',
  'application/vnd.openxmlformats-officedocument.wordprocessingml.document': 'docx',
  'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet': 'xlsx',
  'application/pdf': 'pdf',
};

/** 按扩展名优先、mime 兜底识别类型；未识别返回 null。 */
export function detectType(filename: string, mime?: string): DocType | null {
  const dot = filename.lastIndexOf('.');
  if (dot >= 0) {
    const byExt = EXT[filename.slice(dot + 1).toLowerCase()];
    if (byExt) return byExt;
  }
  if (mime) return MIME[mime.toLowerCase()] ?? null;
  return null;
}

export async function parseDocument(input: ParseInput, deps: ParseDeps = {}): Promise<string> {
  const type = detectType(input.filename, input.mime);
  if (!type) throw new Error('unsupported_type');
  switch (type) {
    case 'txt':
    case 'md':
    case 'csv':
      return input.data.toString('utf8').replace(/^\uFEFF/, '');
    case 'docx':
      return (await mammoth.extractRawText({ buffer: input.data })).value;
    case 'xlsx':
      return parseXlsx(input.data);
    case 'pdf':
      return (deps.pdf ?? parsePdf)(input.data);
  }
}

/** 逐表逐行抽文本，单元格以制表符连接；公式取 result，富文本取 text。 */
export async function parseXlsx(data: Buffer): Promise<string> {
  const wb = new ExcelJS.Workbook();
  await wb.xlsx.load(data);
  const lines: string[] = [];
  wb.eachSheet((ws) => {
    ws.eachRow((row) => {
      const cells = (row.values as unknown[]).slice(1).map(cellText);
      lines.push(cells.join('\t'));
    });
  });
  return lines.join('\n');
}

function cellText(v: unknown): string {
  if (v === null || v === undefined) return '';
  if (typeof v !== 'object') return String(v);
  const o = v as { text?: unknown; result?: unknown };
  if (o.text !== undefined) return String(o.text);
  if (o.result !== undefined) return String(o.result ?? '');
  return '';
}

/** pdfjs 的 legacy build：逐页取文本项拼接（文本抽取不需要字体/画布）。 */
async function parsePdf(data: Buffer): Promise<string> {
  const pdfjs = await import('pdfjs-dist/legacy/build/pdf.mjs');
  const doc = await pdfjs.getDocument({ data: new Uint8Array(data), isEvalSupported: false }).promise;
  const parts: string[] = [];
  for (let i = 1; i <= doc.numPages; i++) {
    const page = await doc.getPage(i);
    const content = await page.getTextContent();
    parts.push(content.items.map((it) => it.str ?? '').join(' '));
  }
  return parts.join('\n');
}
```

```ts
/** 归一化：统一换行、压缩行内空白、去掉多余空行并 trim 两端。 */
export function normalizeText(raw: string): string {
  return raw
    .replace(/\r\n?/g, '\n')
    .replace(/[ \t\f\v]+/g, ' ')
    .replace(/ *\n */g, '\n')
    .replace(/\n{3,}/g, '\n\n')
    .trim();
}

/**
 * 字符滑窗切分：窗口尾部若落在行中，就近回退到换行边界（不早于窗口一半）以少切碎行，再 trim。
 * 下一窗口起点以真实 end 为锚（end - overlap），故换行回退超过 overlap 时不会漏字；
 * overlap≥size 时用 max(start+1, …) 兜底为 1，避免死循环。
 */
export function chunkText(raw: string, size = 800, overlap = 150): string[] {
  const text = normalizeText(raw);
  if (!text) return [];
  const chunks: string[] = [];
  let start = 0;
  while (start < text.length) {
    let end = Math.min(start + size, text.length);
    if (end < text.length) {
      const boundary = text.lastIndexOf('\n', end - 1);
      if (boundary > start + Math.floor(size / 2)) end = boundary;
    }
    const piece = text.slice(start, end).trim();
    if (piece) chunks.push(piece);
    if (end >= text.length) break;
    // 关键：以真实 end 锚定下一窗口起点；固定步进会在换行回退处留下缝隙导致漏字。
    start = Math.max(start + 1, end - overlap);
  }
  return chunks;
}
```

```ts
import type { EmbeddingGateway } from '../embedding/types';
import { parseDocument, type ParseDeps, type ParseInput } from './parse';
import { chunkText } from './chunk';

export interface IngestDeps extends ParseDeps {
  embed: EmbeddingGateway;
}

export interface IngestOptions {
  chunkSize?: number;
  chunkOverlap?: number;
}

export interface IngestedChunk {
  seq: number;
  text: string;
  vector: number[];
}

/** 编排：解析出文本 → 切分 → 批量嵌入 → 组装带 seq 的切片。空文档视为失败。 */
export async function ingest(
  input: ParseInput,
  deps: IngestDeps,
  opts: IngestOptions = {},
): Promise<IngestedChunk[]> {
  const text = await parseDocument(input, deps);
  if (!text.trim()) throw new Error('empty_document');
  const pieces = chunkText(text, opts.chunkSize ?? 800, opts.chunkOverlap ?? 150);
  if (pieces.length === 0) throw new Error('empty_document');
  const vectors = await deps.embed.embed(pieces);
  return pieces.map((t, i) => ({ seq: i, text: t, vector: vectors[i] }));
}
```

```ts
export async function searchKb(cfg: KbClientConfig, params: KbSearchParams): Promise<KbSearchHit[]> {

    body: JSON.stringify(params),
```

```ts
import type { KbSearchHit } from '@ai-office/shared';
import type { Tool } from './types';
import type { EmbeddingGateway } from '../embedding/types';
import { searchKb, type KbClientConfig } from '../kb/bff-client';

const MAX_SNIPPET = 200;

export interface KbRetrieveConfig {
  baseUrl: string;
  internalToken?: string;
  embed: EmbeddingGateway;
  fetchImpl?: typeof fetch;
}

/** 检索用户知识库并返回带来源引用的切片。safe：只读、无副作用。 */
export function createKbRetrieveTool(cfg: KbRetrieveConfig): Tool {
  const client: KbClientConfig = {
    baseUrl: cfg.baseUrl,
    internalToken: cfg.internalToken,
    fetchImpl: cfg.fetchImpl,
  };
  return {
    name: 'kb.retrieve',
    description:
      '检索用户的知识库，返回与查询最相关的文档切片及来源引用。当用户问题可能涉及已上传的私有文档时使用。',
    parameters: {
      type: 'object',
      properties: {
        query: { type: 'string', description: '检索查询（自然语言）' },
        topK: { type: 'number', description: '返回条数，1-20，默认 5' },
      },
      required: ['query'],
    },
    permission: 'safe',
    async execute(args, ctx) {
      const query = String(args.query ?? '').trim();
      const topK = clampTopK(args.topK);
      let hits: KbSearchHit[];
      try {
        const [vector] = await cfg.embed.embed([query]);
        hits = await searchKb(client, {
          vector,
          topK,
          tenantId: ctx.tenantId,
          userId: ctx.userId,
        });
      } catch {
        // 检索不可达不中断 agent loop：回结构化失败，模型据此说明。
        return {
          ok: false,
          tool: 'kb.retrieve',
          error: 'kb_search_unreachable',
          text: '知识库暂时不可用',
          display: { sources: [] },
        };
      }
      if (hits.length === 0) {
        return { ok: true, tool: 'kb.retrieve', text: '知识库无匹配内容', display: { sources: [] } };
      }
      return {
        ok: true,
        tool: 'kb.retrieve',
        text: hits.map((h) => `[来源: ${h.docName}#${h.seq}] ${h.text}`).join('\n\n'),
        display: {
          sources: hits.map((h) => ({
            docName: h.docName,
            seq: h.seq,
            score: h.score,
            snippet: h.text.slice(0, MAX_SNIPPET),
          })),
        },
      };
    },
  };
}

/** clamp 到 [1,20]，默认 5；非数字回退默认。 */
function clampTopK(v: unknown): number {
  const n = typeof v === 'number' ? v : Number(v);
  if (!Number.isFinite(n)) return 5;
  return Math.min(20, Math.max(1, Math.trunc(n)));
}
```

```ts
import { randomUUID } from 'node:crypto';
import { Injectable, Logger, NotFoundException } from '@nestjs/common';
import type { KbDocDto, KbSearchHit, KbDocStatus } from '@ai-office/shared';
import { PrismaService } from '../db/prisma.service';
import type { KbStorage } from './storage';

export interface KbIngestConfig {
  runtimeUrl?: string;
  runtimeToken?: string;
  fetchImpl?: typeof fetch;
}

export interface CreateDocInput {
  tenantId: string;
  userId: string;
  filename: string;
  mime: string;
  size: number;
  storageKey: string;
}

/** 摄取所需的最小文档投影（避免把整行塞进编排）。 */
export type IngestTarget = Pick<
  CreateDocInput,
  'tenantId' | 'userId' | 'filename' | 'mime' | 'storageKey'
> & { id: string };

@Injectable()
export class KbService {
  private readonly logger = new Logger(KbService.name);

  constructor(
    private readonly prisma: PrismaService,
    private readonly storage: KbStorage,
    private readonly cfg: KbIngestConfig = {},
  ) {}

  async create(input: CreateDocInput): Promise<KbDocDto> {
    const row = await this.prisma.kbDoc.create({
      data: {
        tenantId: input.tenantId,
        userId: input.userId,
        filename: input.filename,
        mime: input.mime,
        size: input.size,
        storageKey: input.storageKey,
        status: 'pending',
      },
    });
    return toDto(row);
  }

  async list(tenantId: string, userId: string): Promise<KbDocDto[]> {
    const rows = await this.prisma.kbDoc.findMany({
      where: { tenantId, userId },
      orderBy: { createdAt: 'desc' },
    });
    return rows.map(toDto);
  }

  async remove(id: string, tenantId: string, userId: string): Promise<void> {
    const doc = await this.prisma.kbDoc.findFirst({ where: { id, tenantId, userId } });
    if (!doc) throw new NotFoundException('doc_not_found');
    // 级联删除切片由外键 onDelete: Cascade 承担；显式删 doc 即可。
    await this.prisma.kbDoc.delete({ where: { id } });
  }

  /** 向量检索：按 tenant+user 过滤，HNSW 余弦距离升序取 topK；相似度 = 1 - 距离。 */
  async search(
    tenantId: string,
    userId: string,
    vector: number[],
    topK: number,
  ): Promise<KbSearchHit[]> {
    const lit = toVectorLiteral(vector);
    const rows = await this.prisma.$queryRaw<
      { docName: string; seq: number; dist: number; text: string }[]
    >`
      SELECT d."filename" AS "docName", c."seq", c."text", (c."vector" <=> ${lit}::vector) AS "dist"
      FROM "KbChunk" c JOIN "KbDoc" d ON d."id" = c."docId"
      WHERE c."tenantId" = ${tenantId} AND c."userId" = ${userId}
      ORDER BY c."vector" <=> ${lit}::vector
      LIMIT ${topK}
    `;
    return rows.map((r) => ({
      docName: r.docName,
      seq: Number(r.seq),
      score: 1 - Number(r.dist),
      text: r.text,
    }));
  }

  /**
   * 摄取编排：调 runtime /kb/ingest 取切片（含向量），写 KbChunk（向量走原始 SQL），
   * 并把 doc 置 ready。任一步失败抛错，由调用方置 failed。
   */
  async ingest(target: IngestTarget): Promise<void> {
    const doFetch = this.cfg.fetchImpl ?? fetch;
    const base = this.cfg.runtimeUrl ?? process.env.AGENT_RUNTIME_URL ?? 'http://localhost:4001';
    const token = this.cfg.runtimeToken ?? process.env.AGENT_RUNTIME_TOKEN;
    let res: Response;
    try {
      res = await doFetch(`${base}/kb/ingest`, {
        method: 'POST',
        headers: {
          'content-type': 'application/json',
          ...(token ? { 'x-agent-runtime-token': token } : {}),
        },
        body: JSON.stringify({
          storageKey: target.storageKey,
          filename: target.filename,
          mime: target.mime,
        }),
      });
    } catch {
      throw new Error('runtime_unreachable');
    }
    if (!res.ok) throw new Error(`runtime_http_${res.status}`);
    const data = (await res.json()) as {
      ok: boolean;
      error?: string;
      chunks?: { seq: number; text: string; vector: number[] }[];
    };
    if (!data.ok) throw new Error(data.error ?? 'ingest_failed');
    const chunks = data.chunks ?? [];

    for (const c of chunks) {
      await this.prisma.$executeRaw`
        INSERT INTO "KbChunk" ("id", "docId", "tenantId", "userId", "seq", "text", "vector", "createdAt")
        VALUES (${randomUUID()}, ${target.id}, ${target.tenantId}, ${target.userId}, ${c.seq}, ${c.text}, ${toVectorLiteral(c.vector)}::vector, NOW())
      `;
    }
    await this.prisma.kbDoc.update({
      where: { id: target.id },
      data: { status: 'ready', chunkCount: chunks.length, error: null },
    });
  }

  /** 摄取失败收尾：置 failed + error（best-effort，落库失败不再抛）。 */
  async markFailed(id: string, error: string): Promise<void> {
    try {
      await this.prisma.kbDoc.update({ where: { id }, data: { status: 'failed', error } });
    } catch (err) {
      this.logger.error(`kb doc ${id} 置 failed 失败: ${(err as Error).message}`);
    }
  }
}

function toVectorLiteral(v: number[]): string {
  return `[${v.join(',')}]`;
}

function toDto(row: {
  id: string;
  filename: string;
  mime: string;
  size: number;
  status: string;
  chunkCount: number;
  error: string | null;
  createdAt: Date;
}): KbDocDto {
  return {
    id: row.id,
    filename: row.filename,
    mime: row.mime,
    size: row.size,
    status: row.status as KbDocStatus,
    chunkCount: row.chunkCount,
    error: row.error,
    createdAt: row.createdAt.toISOString(),
  };
}
```

```ts
import { randomUUID } from 'node:crypto';
import {
  Controller,
  Delete,
  Get,
  HttpCode,
  Logger,
  Param,
  Post,
  Req,
  Res,
  UseGuards,
} from '@nestjs/common';
import type { FastifyRequest, FastifyReply } from 'fastify';
import type { MultipartFile } from '@fastify/multipart';
import type { KbDocDto } from '@ai-office/shared';
import { KbService } from './kb.service';
import { KB_STORAGE, type KbStorage } from './storage';
import { Inject } from '@nestjs/common';
import { JwtGuard } from '../auth/jwt.guard';
import { CurrentUser } from '../auth/current-user';
import type { JwtPayload } from '../auth/jwt';

const ALLOWED_EXT = ['txt', 'md', 'csv', 'docx', 'xlsx', 'pdf'];

@Controller('api/kb/docs')
@UseGuards(JwtGuard)
export class KbController {
  private readonly logger = new Logger(KbController.name);

  constructor(
    private readonly kb: KbService,
    @Inject(KB_STORAGE) private readonly storage: KbStorage,
  ) {}

  @Get()
  list(@CurrentUser() user: JwtPayload): Promise<KbDocDto[]> {
    return this.kb.list(user.tenantId, user.sub);
  }

  @Post()
  async upload(
    @CurrentUser() user: JwtPayload,
    @Req() req: FastifyRequest,
    @Res() reply: FastifyReply,
  ): Promise<void> {
    let file: MultipartFile | undefined;
    try {
      file = await req.file();
    } catch {
      // @fastify/multipart 在超限时抛错（RequestFileTooLargeError 等）。
      reply.status(413).send({ error: 'file_too_large' });
      return;
    }
    if (!file) {
      reply.status(400).send({ error: 'missing_file' });
      return;
    }
    const ext = file.filename.includes('.') ? file.filename.split('.').pop()!.toLowerCase() : '';
    if (!ALLOWED_EXT.includes(ext)) {
      // 需要排空流，否则连接悬挂。
      await file.toBuffer().catch(() => undefined);
      reply.status(400).send({ error: 'unsupported_type' });
      return;
    }
    let buffer: Buffer;
    try {
      buffer = await file.toBuffer();
    } catch {
      reply.status(413).send({ error: 'file_too_large' });
      return;
    }
    const mime = file.mimetype || 'application/octet-stream';
    const storageKey = `kb/${user.tenantId}/${user.sub}/${randomUUID()}/${file.filename}`;
    await this.storage.putObject(storageKey, buffer, mime);
    const doc = await this.kb.create({
      tenantId: user.tenantId,
      userId: user.sub,
      filename: file.filename,
      mime,
      size: buffer.length,
      storageKey,
    });
    // 异步摄取：不阻塞上传响应，前端轮询 status；失败置 failed + error。
    void this.kb
      .ingest({ id: doc.id, tenantId: user.tenantId, userId: user.sub, filename: doc.filename, mime: doc.mime, storageKey })
      .catch(async (err: unknown) => {
        const msg = err instanceof Error ? err.message : String(err);
        this.logger.warn(`kb doc ${doc.id} 摄取失败: ${msg}`);
        await this.kb.markFailed(doc.id, msg);
      });
    reply.status(202).send(doc);
  }

  @Delete(':id')
  @HttpCode(204)
  async remove(
    @CurrentUser() user: JwtPayload,
```

```ts
import { Body, Controller, Post, Req, Res } from '@nestjs/common';
import type { FastifyRequest, FastifyReply } from 'fastify';
import type { KbSearchHit } from '@ai-office/shared';
import { KbService } from './kb.service';

interface SearchBody {
  vector?: number[];
  topK?: number;
  tenantId?: string;
  userId?: string;
}

/** 内部检索端点：供 runtime 调用（不是浏览器）。用共享密钥 INTERNAL_TOKEN 校验，空则不校验（仅限本地）。 */
@Controller('internal/kb')
export class InternalKbController {
  constructor(private readonly kb: KbService) {}

  @Post('search')
  async search(
    @Req() req: FastifyRequest,
    @Body() body: SearchBody,
    @Res() reply: FastifyReply,
  ): Promise<void> {
    const token = process.env.INTERNAL_TOKEN;
    if (token && req.headers['x-internal-token'] !== token) {
      reply.status(401).send({ error: 'unauthorized' });
      return;
    }
    const { vector, topK, tenantId, userId } = body ?? {};
    if (!Array.isArray(vector) || !tenantId || !userId) {
      reply.status(400).send({ error: 'invalid_params' });
      return;
    }
    const hits: KbSearchHit[] = await this.kb.search(
      tenantId,
      userId,
      vector,
      clampTopK(topK),
    );
    reply.status(200).send(hits);
  }
}

function clampTopK(v: unknown): number {
  const n = typeof v === 'number' ? v : Number(v);
  if (!Number.isFinite(n)) return 5;
  return Math.min(20, Math.max(1, Math.trunc(n)));
}
```

```ts
import type { KbDocDto } from '@ai-office/shared';

export async function listKbDocs(): Promise<KbDocDto[]> {
  const res = await authFetch('/api/kb/docs');
  if (!res.ok) throw new Error(`kb_list_failed_${res.status}`);
  return (await res.json()) as KbDocDto[];
}

/** 上传知识库文档。用 FormData：**不能**设 content-type，否则 boundary 丢失（故不走 authFetch 的默认头）。 */
export async function uploadKbDoc(file: File): Promise<KbDocDto> {
  const token = getToken();
  const form = new FormData();
  form.append('file', file);
  const res = await fetch(`${BFF_URL}/api/kb/docs`, {
    method: 'POST',
    headers: token ? { authorization: `Bearer ${token}` } : {},
    body: form,
  });
  if (res.status === 401) {
    clearToken();
    throw new Error('unauthorized');
  }
  if (!res.ok) throw new Error(`kb_upload_failed_${res.status}`);
  return (await res.json()) as KbDocDto;
}

export async function deleteKbDoc(id: string): Promise<void> {
  const res = await authFetch(`/api/kb/docs/${id}`, { method: 'DELETE' });
  if (!res.ok) throw new Error(`kb_delete_failed_${res.status}`);
}
```

```tsx
'use client';

import { useCallback, useEffect, useRef, useState } from 'react';
import { useRouter } from 'next/navigation';
import type { KbDocDto } from '@ai-office/shared';
import { deleteKbDoc, getToken, isUnauthorized, listKbDocs, uploadKbDoc } from '../../lib/api';

export default function KbPage() {
  const router = useRouter();
  const [docs, setDocs] = useState<KbDocDto[]>([]);
  const [error, setError] = useState('');
  const [busy, setBusy] = useState(false);
  const fileRef = useRef<HTMLInputElement>(null);

  const refresh = useCallback(async () => {
    try {
      setDocs(await listKbDocs());
    } catch (err) {
      if (isUnauthorized(err)) {
        router.push('/login');
        return;
      }
      setError(String(err));
    }
  }, [router]);

  useEffect(() => {
    if (!getToken()) {
      router.push('/login');
      return;
    }
    void refresh();
  }, [refresh, router]);

  // 存在 pending 项时轮询刷新，全部落定后停止。
  useEffect(() => {
    if (!docs.some((d) => d.status === 'pending')) return;
    const t = setInterval(() => void refresh(), 2500);
    return () => clearInterval(t);
  }, [docs, refresh]);

  async function upload() {
    const file = fileRef.current?.files?.[0];
    if (!file) return;
    setBusy(true);
    setError('');
    try {
      const doc = await uploadKbDoc(file);
      setDocs((prev) => [doc, ...prev]);
      if (fileRef.current) fileRef.current.value = '';
    } catch (err) {
      if (isUnauthorized(err)) {
        router.push('/login');
        return;
      }
      setError(String(err));
    } finally {
      setBusy(false);
    }
  }

  async function remove(id: string) {
    try {
      await deleteKbDoc(id);
      setDocs((prev) => prev.filter((d) => d.id !== id));
    } catch (err) {
      if (isUnauthorized(err)) {
        router.push('/login');
        return;
      }
      setError(String(err));
    }
  }

  const badge: Record<KbDocDto['status'], string> = { pending: '摄取中', ready: '就绪', failed: '失败' };

  return (
    <main>
      <h1>知识库</h1>
      <div style={{ display: 'flex', gap: 8, marginBottom: 16 }}>
        <input ref={fileRef} type="file" accept=".txt,.md,.csv,.docx,.xlsx,.pdf" />
        <button onClick={() => void upload()} disabled={busy}>
          {busy ? '上传中…' : '上传'}
        </button>
        <button onClick={() => router.push('/')} style={{ marginLeft: 'auto' }}>
          返回工作台
        </button>
      </div>
      {error && <p style={{ color: 'crimson' }}>{error}</p>}
      {docs.length === 0 ? (
        <p style={{ color: '#888' }}>暂无文档</p>
      ) : (
        <ul style={{ listStyle: 'none', padding: 0 }}>
          {docs.map((d) => (
            <li key={d.id} style={{ border: '1px solid #ddd', borderRadius: 6, padding: 12, marginBottom: 8 }}>
              <div style={{ fontWeight: 600 }}>{d.filename}</div>
              <div style={{ color: '#666', fontSize: 13 }}>
                {badge[d.status]} · {d.chunkCount} 切片 · {new Date(d.createdAt).toLocaleString()}
                {d.error ? ` · ${d.error}` : ''}
              </div>
              <button style={{ marginTop: 8 }} onClick={() => void remove(d.id)}>
                删除
              </button>
            </li>
          ))}
        </ul>
      )}
    </main>
  );
}
```
